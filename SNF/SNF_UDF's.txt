CREATE OR REPLACE FUNCTION REPORTING_DB.EDW.UDF_GET_COUNTRY_ISO("LAT" FLOAT, "LON" FLOAT)
RETURNS VARCHAR(16777216)
LANGUAGE SQL
COMMENT='Please pass in a latitude coordinate and a longitude coordinate and I will return a Country ISO value.'
AS ' 
    select 
      dt.COUNTRY_ISO
    from 
        (select 
            COUNTRY_ISO, 
            st_contains(POLYGON, 
                        -- This looks reversed but it is correct
                        TO_GEOGRAPHY(''POINT('' || cast( lon as varchar(16777216)) || '' '' || cast(lat as varchar(16777216)) || '')'' )) as INCLUDES 
          from 
            REPORTING_DB.EDW_TABLES.LOOKUP_COUNTRY
        ) dt 
     where 
        dt.INCLUDES = true 
  ';



CREATE OR REPLACE FUNCTION REPORTING_DB.EDW.UDF_VARCHAR_TO_DATE("COL_NAME" VARCHAR(16777216), "FORMAT_DATE" VARCHAR(16777216))
RETURNS DATE
LANGUAGE SQL
COMMENT='This is useful for turning varchar dates (SAP) into date data type as it handles for the 0000000 case'
AS '
    SELECT
    	-- If it is all 0s return null
    	CASE
        	WHEN col_name = REPEAT(0, (SELECT length(format_date))) THEN NULL
        	WHEN TRIM(col_name) = '''' THEN NULL
        	ELSE TO_DATE(col_name, format_date)
        END
    ';
